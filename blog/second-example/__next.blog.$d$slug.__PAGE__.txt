def train_step(x, y, model, cfg: Config):
    pred = model(x)
    loss = ((pred - y) ** 2).mean()
    return loss

$$
\begin{aligned}
\text{MSE} &= \frac{1}{N}\sum_{i=1}^N (y_i - \hat{y}*i)^2 \

\end{aligned}
